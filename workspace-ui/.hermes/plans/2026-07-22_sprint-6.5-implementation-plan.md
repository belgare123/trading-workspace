### Task 1.2: Create EventEnvelope.ts

```typescript
// src/event-journal/EventEnvelope.ts

import { randomUUID } from 'crypto'
import type { RuntimeId } from './types'

export interface EventEnvelopeMetadata {
  version: number
  schemaVersion: number
  source: string
}

export interface EventEnvelope<T = unknown> {
  id: string
  traceId: string
  causationId?: string
  correlationId?: string
  tradeId?: string
  runtime: RuntimeId
  type: string
  sequence: number
  timestamp: number
  payload: T
  metadata: EventEnvelopeMetadata
}

export interface CreateEventEnvelopeInput<T = unknown> {
  traceId: string
  causationId?: string
  correlationId?: string
  tradeId?: string
  runtime: RuntimeId
  type: string
  payload: T
  source?: string
  metadata?: Partial<EventEnvelopeMetadata>
}

/** Build a new EventEnvelope. Sequence is set by the Journal. */
export function createEventEnvelope<T>(input: CreateEventEnvelopeInput<T>): Omit<EventEnvelope<T>, 'sequence'> {
  return {
    id: randomUUID(),
    traceId: input.traceId,
    causationId: input.causationId,
    correlationId: input.correlationId,
    tradeId: input.tradeId,
    runtime: input.runtime,
    type: input.type,
    sequence: 0, // placeholder — set by journal
    timestamp: Date.now(),
    payload: input.payload,
    metadata: {
      version: input.metadata?.version ?? 1,
      schemaVersion: input.metadata?.schemaVersion ?? 1,
      source: input.source ?? 'workspace-ui:2.0.0',
    },
  }
}
```

### Task 3.1: Install better-sqlite3 types + SQLite event journal

Key implementation points:
- PRAGMA journal_mode=WAL, synchronous=NORMAL, busy_timeout=5000
- `initTables()`: CREATE TABLE IF NOT EXISTS events + snapshots
- `append()`: INSERT into events, return complete envelope with sequence (AUTOINCREMENT)
- `flush()`: no-op for SQLite (each INSERT is synchronous by default)
- `read()`: SELECT with optional WHERE clauses for each filter field
- `recover()`: verify integrity, return counts
- `getCurrentSequence()`: SELECT MAX(sequence) FROM events

AsyncIterable for read() — use cursor-based approach with `better-sqlite3`'s `iterate()` method wrapped in an async generator.

### Task 3.2: Test SQLiteEventJournal

Test cases:
- Append single event, verify sequence auto-increment
- Append multiple events, verify read returns them in order
- Read with filter (by tradeId, traceId, runtime, type)

- Recover after events appended
- Close and verify no further operations

```typescript
export interface JournalWriterConfig {
  journal: IEventJournal
  flushConfig: JournalFlushConfig
  onCriticalEvent?: (envelope: EventEnvelope) => void
}

export class JournalWriter {
  private buffer: Omit<EventEnvelope, 'sequence'>[] = []
  private dirty = false
  private timer?: ReturnType<typeof setTimeout>

  constructor(private config: JournalWriterConfig) {}

  async append<T>(input: CreateEventEnvelopeInput<T>): Promise<EventEnvelope>
  // Appends through the journal.
  // - If critical event type: append immediately + fsync via journal.flush()
  // - If non-critical: buffer.append(). If buffer.length >= maxBufferedEvents → flush batch
  private startTimer(): void
  // Starts interval timer. On tick if dirty → flush
  async flush(): Promise<void>
  // Flush buffer + journal.flush()
  async shutdown(): Promise<void>
  // Flush + close
  isCritical(type: string): boolean
}
```

### Task 7.1: StateHasher

```typescript
import { createHash } from 'crypto'

export function hashObject(obj: unknown): string {
  return createHash('sha256')
    .update(JSON.stringify(obj, Object.keys(obj).sort()))
    .digest('hex')
}

export function hashState(state: Record<string, unknown>): string {
  // Sort keys for deterministic ordering
  const sorted: Record<string, unknown> = {}
  for (const key of Object.keys(state).sort()) {
    sorted[key] = state[key]
  }
  return hashObject(sorted)
}
```

## Risks & Mitigations

| Risk | Mitigation |
|------|-----------|
| better-sqlite3 native compilation fails on Windows | Install Visual Studio Build Tools or use prebuilt binary. Fallback: sql.js (WASM) |
| AsyncIterable not supported in target | TypeScript 6.0 + Node.js 20+ supports it natively. Use `for await...of` |
| Snapshot state serialization size | Limit state to ~10KB max per snapshot, compress with gzip if needed |
| SQLite concurrent access | WAL mode handles reads during writes. Single-process architecture avoids contention |
| UUIDv7 not in Node.js crypto | Use `uuid` package with `v7()` or `randomUUID()` with custom timestamp prefix |
